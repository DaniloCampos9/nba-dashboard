def atualizar_elencos_via_espn():
    print("--- INICIANDO RASTREAMENTO DE ELENCOS VIA ESPN API ---")
    
    # 1. Puxar todos os jogadores do nosso Supabase para bater os nomes
    print("Baixando base de jogadores do Supabase...")
    resp_jog = supabase.table("jogadores").select("id, nome_completo").execute()
    df_nossos_jogadores = pd.DataFrame(resp_jog.data)
    
    if df_nossos_jogadores.empty:
        print("Erro: Nenhum jogador no banco.")
        return

    # Vamos converter os nomes para minúsculas para facilitar o cruzamento
    df_nossos_jogadores['nome_lower'] = df_nossos_jogadores['nome_completo'].str.lower()
    
    # 2. Obter os times da ESPN
    url_espn_teams = "http://site.api.espn.com/apis/site/v2/sports/basketball/nba/teams?limit=30"
    req = requests.get(url_espn_teams)
    times_espn = req.json()['sports'][0]['leagues'][0]['teams']
    
    atualizacoes_cadastrais = []
    jogadores_nao_encontrados = []

    for item in times_espn:
        time_espn = item['team']
        abrev = time_espn['abbreviation'] # Ex: LAL, BOS
        espn_team_id = time_espn['id']
        nome_time = time_espn['displayName']
        
        # Achar o ID oficial da NBA para esse time usando nossa sigla
        nba_team_info = teams.find_team_by_abbreviation(abrev)
        if not nba_team_info:
            continue
            
        nba_team_id = nba_team_info['id']
        print(f"Lendo elenco: {nome_time} ({abrev})")
        
        # 3. Puxar o roster específico desse time na ESPN
        url_roster = f"http://site.api.espn.com/apis/site/v2/sports/basketball/nba/teams/{espn_team_id}/roster"
        req_roster = requests.get(url_roster)
        
        try:
            atletas = req_roster.json()['athletes']
            for atleta in atletas:
                nome_espn = atleta['fullName'].lower()
                
                # Procura esse nome na nossa base do Supabase
                match = df_nossos_jogadores[df_nossos_jogadores['nome_lower'] == nome_espn]
                
                if not match.empty:
                    nba_player_id = int(match.iloc[0]['id'])
                    atualizacoes_cadastrais.append({
                        "id": nba_player_id,
                        "time_atual_id": nba_team_id
                    })
                else:
                    # Alguns nomes têm sufixos diferentes (ex: Kelly Oubre Jr. vs Kelly Oubre)
                    jogadores_nao_encontrados.append(f"{atleta['fullName']} ({abrev})")
        except KeyError:
            print(f"Erro ao ler os atletas de {nome_time}")
            
        time.sleep(0.5)

    # 4. Enviar para o Supabase
    if atualizacoes_cadastrais:
        print(f"\nSalvando {len(atualizacoes_cadastrais)} jogadores cruzados no Supabase...")
        from carregador import dividir_em_lotes
        for lote in dividir_em_lotes(atualizacoes_cadastrais, 100):
            supabase.table("jogadores").upsert(lote).execute()
        print("✅ ELENCOS ATUALIZADOS COM SUCESSO PELA ESPN!")
        
    if jogadores_nao_encontrados:
        print("\n⚠️ Jogadores ignorados por divergência de nome (Jr., Sr., hífens):")
        for j in jogadores_nao_encontrados[:15]:
            print(f" - {j}")
        print(f"... e mais {len(jogadores_nao_encontrados) - 15} outros.")

if __name__ == "__main__":
    atualizar_elencos_via_espn()

            except Exception as e:
                tentativas_carreira += 1
                if tentativas_carreira < 3:
                    print(f"  ⏳ Tropeço na carreira. Respirando 5s para tentar de novo...")
                    time.sleep(5)

                                path=caminho_arquivo,

                            )
                            print(f"  🏀 Shot Chart {season} ({tipo_temp}) salvo no Storage.")
                        
                        break # Se chegou aqui sem erro (com ou sem arremessos), sai do loop de tentativas
                        
                    except Exception as e:
                        erro_str = str(e)
                        # Se for erro de JSON Vazio (Bloqueio) ou Timeout, nós tentamos de novo

                        else:
                            # Se for outro tipo de erro (ex: jogador não jogou na liga ainda), ignora e segue
                            break

def dividir_em_lotes(lista, tamanho=100):
    """Fatia listas gigantes em pedaços menores para o banco de dados não engasgar."""
    for i in range(0, len(lista), tamanho):
        yield lista[i:i + tamanho]

@retry(stop=stop_after_attempt(3), wait=wait_fixed(5))

    if not dados_limpos: return

def extrair_dados_defesa(temporada_nba, tipo="Regular Season"):
    dados_trad = leaguedashplayerstats.LeagueDashPlayerStats(season=temporada_nba, season_type_all_star=tipo).get_data_frames()[0]
    df_trad = dados_trad[['PLAYER_ID', 'STL', 'BLK']]
    
    dados_hustle = leaguehustlestatsplayer.LeagueHustleStatsPlayer(season=temporada_nba, season_type_all_star=tipo).get_data_frames()[0]
    df_hustle = dados_hustle[['PLAYER_ID', 'DEFLECTIONS']]
    
    df_defesa = pd.merge(df_trad, df_hustle, on='PLAYER_ID', how='inner')
    df_defesa['temporada'] = temporada_nba
    df_defesa['tipo_temporada'] = tipo
    return df_defesa

def extrair_dados_times(temporada_nba, tipo="Regular Season"):
    try:
        dados_times = leaguedashteamstats.LeagueDashTeamStats(
            season=temporada_nba, measure_type_detailed_defense='Advanced', season_type_all_star=tipo
        ).get_data_frames()[0]
        
        df_times = dados_times[['TEAM_ID', 'TEAM_NAME', 'GP', 'W', 'L', 'OFF_RATING', 'DEF_RATING', 'NET_RATING', 'PACE']].copy()
        df_times['temporada'] = temporada_nba
        df_times['tipo_temporada'] = tipo
        return df_times
    except Exception as e:
        print(f"Erro na extração de times: {e}")
        return pd.DataFrame()

def executar_atualizacao_diaria(temporada_atual="2026-27"):
    """Rotina ultrarrápida otimizada para rodar diariamente no PC."""
    print(f"--- INICIANDO ATUALIZAÇÃO DIÁRIA DA NBA ({temporada_atual}) ---")
    
    ids_jogadores = obter_ids_jogadores_ativos()
    limpar_dados_temporada_atual(temporada_atual)
    
    try:
        print("Baixando Estatísticas Tradicionais (Incluindo 3 Pontos)...")

        df_trad = leaguedashplayerstats.LeagueDashPlayerStats(season=temporada_atual, season_type_all_star=tipo).get_data_frames()[0]
        
        lista_trad = []
        for _, row in df_trad.iterrows():
            if row['PLAYER_ID'] in ids_jogadores:
                lista_trad.append({

                    "jogos_disputados": int(row['GP']),
                    "minutos_totais": float(row['MIN']),
                    "pontos_totais": int(row['PTS']),
                    "assistencias_totais": int(row['AST']),
                    "rebotes_totais": int(row['REB']),
                    "fg3m_totais": int(row['FG3M']),
                    "fg3a_totais": int(row['FG3A'])
                })
        
        if lista_trad: enviar_estatisticas_para_banco(lista_trad)
        
        bruto_adv, _ = buscar_estatisticas_avancadas_liga(temporada_atual)

        insidetextfont=dict(color='white', size=14, family="Arial Black")

    avg_def = df['def_rating'].mean()

    fig = px.scatter(
        df,
        x='off_rating',
        y='def_rating',
        text='nome_time',
        hover_data=['vitorias', 'derrotas', 'net_rating', 'pace'],

        },
        title='League Landscape: Identidade e Eficiência'
    )

    # O SEGREDO ANALÍTICO: Inverter o eixo Y. 
    # Em Def Rating, números menores são melhores (sofreu menos pontos).
    # Invertendo, a melhor defesa fica no TOPO do gráfico.
    fig.update_yaxes(autorange="reversed")

    fig.add_hline(y=avg_def, line_dash="dash", line_color="gray", opacity=0.5)
    fig.add_vline(x=avg_off, line_dash="dash", line_color="gray", opacity=0.5)

def grafico_radar_comparativo(atletas_validos):
    """
    Gera um Spider Chart (Radar) normalizado comparando o DNA de até 3 jogadores.
    """
    if not atletas_validos:
        return None
        
    fig = go.Figure()
    
    # Paleta de cores moderna: Azul NBA, Vermelho Alerta e Dourado
    cores_linha = ['#1D428A', '#FF4B4B', '#FFD700'] 
    cores_fundo = ['rgba(29, 66, 138, 0.3)', 'rgba(255, 75, 75, 0.3)', 'rgba(255, 215, 0, 0.3)']
    
    # Os 6 eixos do nosso DNA analítico
    categorias = ['Pontuação (PPG)', 'Rebotes (RPG)', 'Playmaking (APG)', 
                  'Eficiência (TS%)', 'Volume (USG%)', 'Impacto (PIE)']
                  
    for idx, a in enumerate(atletas_validos):
        stats = a["stats"]
        jogos = stats.get("jogos_disputados", 1)
        if jogos == 0: jogos = 1
        
        # Extração bruta
        ppg = stats.get("pontos_totais", 0) / jogos
        rpg = stats.get("rebotes_totais", 0) / jogos
        apg = stats.get("assistencias_totais", 0) / jogos
        ts = (stats.get("ts_pct") or 0) * 100
        usg = (stats.get("usg_pct") or 0) * 100
        pie = (stats.get("pie") or 0) * 100
        
        # Normalização matemática (0 a 100) baseada em tetos realistas da NBA
        # (ex: 35 pts = nota 100 em Pontuação)
        valores_normalizados = [
            min((ppg / 35) * 100, 100),
            min((rpg / 15) * 100, 100),
            min((apg / 12) * 100, 100),
            min((ts / 70) * 100, 100),
            min((usg / 40) * 100, 100),
            min((pie / 20) * 100, 100)
        ]
        
        # O Plotly exige que o último valor seja igual ao primeiro para fechar o polígono do radar
        valores_normalizados.append(valores_normalizados[0])
        categorias_fechadas = categorias + [categorias[0]]
        
        fig.add_trace(go.Scatterpolar(
            r=valores_normalizados,
            theta=categorias_fechadas,
            fill='toself',
            name=a["nome"],
            line=dict(color=cores_linha[idx % 3], width=2.5),
            fillcolor=cores_fundo[idx % 3],
            hovertemplate="<b>%{theta}</b><br>Score Relativo: %{r:.1f}/100<extra></extra>"
        ))

    # Estilização visual (Fundo escuro, linhas de grade sutis)
    fig.update_layout(
        polar=dict(
            radialaxis=dict(visible=True, range=[0, 100], showticklabels=False, gridcolor="rgba(255,255,255,0.15)"),
            angularaxis=dict(gridcolor="rgba(255,255,255,0.15)", tickfont=dict(size=12, color="#FAFAFA"))
        ),
        showlegend=True,
        legend=dict(orientation="h", yanchor="bottom", y=-0.2, xanchor="center", x=0.5),
        plot_bgcolor='rgba(0,0,0,0)',
        paper_bgcolor='rgba(0,0,0,0)',
        margin=dict(l=40, r=40, t=40, b=40),
        height=450
    )
    return fig

@st.cache_data(ttl=86400) 
def obter_lista_jogadores():
    try:
        nomes = set()
        inicio = 0
        tamanho_pagina = 1000
        
        while True:
            resp = supabase.table("jogadores").select("nome_completo").range(inicio, inicio + tamanho_pagina - 1).execute()
            
            if not resp.data:
                break
                
            for row in resp.data:
                if row.get('nome_completo'):
                    nomes.add(row['nome_completo'])
            
            if len(resp.data) < tamanho_pagina:
                break
                
            inicio += tamanho_pagina
            
        return sorted(list(nomes))
    except Exception as e:
        print(f"Erro ao carregar lista de jogadores: {e}")
        return []

# 🧠 O CÉREBRO DO FILTRO: Traz tudo se for "Ambos", ou filtra se for específico
def aplicar_filtro(query, tipo):
    if tipo != "Regular Season + Playoffs":
        return query.eq("tipo_temporada", tipo)
    return query

@st.cache_data(ttl=300)
def obter_estatisticas_resumo(jogador_id, temporada, tipo="Regular Season"):
    try:
        q_trad = supabase.table("estatisticas_temporada").select("*").eq("jogador_id", jogador_id).eq("temporada", temporada)
        q_adv = supabase.table("stats_avancadas").select("*").eq("jogador_id", jogador_id).eq("temporada", temporada)
        
        resp_trad = aplicar_filtro(q_trad, tipo).execute()
        resp_adv = aplicar_filtro(q_adv, tipo).execute()
        
        stats = {}
        if resp_trad.data:
            df = pd.DataFrame(resp_trad.data)
            stats['jogos_disputados'] = int(df['jogos_disputados'].sum())
            stats['pontos_totais'] = int(df['pontos_totais'].sum())
            stats['rebotes_totais'] = int(df['rebotes_totais'].sum())
            stats['assistencias_totais'] = int(df['assistencias_totais'].sum())
            stats['fg3m_totais'] = int(df['fg3m_totais'].sum()) if 'fg3m_totais' in df.columns else 0
            stats['fg3a_totais'] = int(df['fg3a_totais'].sum()) if 'fg3a_totais' in df.columns else 0
        if resp_adv.data:
            df = pd.DataFrame(resp_adv.data)
            stats['ts_pct'] = df['ts_pct'].mean()
            stats['usg_pct'] = df['usg_pct'].mean()
            stats['net_rating'] = df['net_rating'].mean()
            stats['off_rating'] = df['off_rating'].mean()
            stats['def_rating'] = df['def_rating'].mean()
            stats['pie'] = df['pie'].mean()
            stats['ast_pct'] = df['ast_pct'].mean()
            stats['pace'] = df['pace'].mean()
        return stats
    except Exception: return {}

@st.cache_data(ttl=3600)
def obter_dados_completos_liga(temporada, tipo="Regular Season"):
    try:
        q_trad = supabase.table("estatisticas_temporada").select("jogador_id, jogos_disputados, pontos_totais, rebotes_totais, assistencias_totais").eq("temporada", temporada)
        q_adv = supabase.table("stats_avancadas").select("jogador_id, ts_pct, usg_pct, net_rating, efg_pct, ast_pct").eq("temporada", temporada)
        
        df_trad = pd.DataFrame(aplicar_filtro(q_trad, tipo).execute().data)
        df_adv = pd.DataFrame(aplicar_filtro(q_adv, tipo).execute().data)
        
        if df_trad.empty or df_adv.empty: return pd.DataFrame()
        
        df_trad = df_trad.groupby("jogador_id").sum().reset_index()
        df_adv = df_adv.groupby("jogador_id").mean().reset_index()
        
        df = pd.merge(df_trad, df_adv, on="jogador_id", how="inner")
        df = df[df['jogos_disputados'] > 0].copy()
        df['PPG'] = df['pontos_totais'] / df['jogos_disputados']
        df['RPG'] = df['rebotes_totais'] / df['jogos_disputados']
        df['APG'] = df['assistencias_totais'] / df['jogos_disputados']
        return df
    except Exception: return pd.DataFrame()

@st.cache_data(ttl=3600)
def obter_dados_graficos_ataque(temporada, tipo="Regular Season"):
    try:
        q_trad = supabase.table("estatisticas_temporada").select("jogador_id, jogos_disputados, pontos_totais").eq("temporada", temporada)
        q_adv = supabase.table("stats_avancadas").select("jogador_id, ts_pct, usg_pct").eq("temporada", temporada)
        df_jog = pd.DataFrame(supabase.table("jogadores").select("id, nome_completo").execute().data)

        df_trad = pd.DataFrame(aplicar_filtro(q_trad, tipo).execute().data)
        df_adv = pd.DataFrame(aplicar_filtro(q_adv, tipo).execute().data)

        if df_trad.empty or df_adv.empty or df_jog.empty: return pd.DataFrame()

        df_trad = df_trad.groupby("jogador_id").sum().reset_index()
        df_adv = df_adv.groupby("jogador_id").mean().reset_index()

        df = pd.merge(df_trad, df_adv, on="jogador_id")
        df = pd.merge(df, df_jog, left_on="jogador_id", right_on="id")

        df = df[df['jogos_disputados'] > 0].copy()
        df['PPG'] = round(df['pontos_totais'] / df['jogos_disputados'], 1)
        df['TS%'] = round(df['ts_pct'] * 100, 1)
        df['USG%'] = round(df['usg_pct'] * 100, 1)
        return df
    except Exception: return pd.DataFrame()

@st.cache_data(ttl=3600)
def obter_mapa_arremessos_jogador(jogador_id, temporada, tipo_temporada="Regular Season"):
    """Busca o mapa de arremessos instantaneamente no Supabase Storage (Buckets)."""
    def buscar_por_tipo(tipo_nba):
        caminho = f"{temporada}/{tipo_nba}/{jogador_id}.parquet"
        try:
            bytes_arquivo = supabase.storage.from_("shotcharts").download(caminho)
            df = pd.read_parquet(io.BytesIO(bytes_arquivo))
            
            # === MÁGICA MATEMÁTICA PARA RECRIAR A COLUNA ===
            # Na NBA, o aro fica no centro (0,0) e cada 10 unidades de LOC equivalem a 1 pé (foot).
            # Usamos Pitágoras (a² + b² = c²) para achar a distância exata do arremesso!
            if not df.empty and 'SHOT_DISTANCE' not in df.columns:
                df['SHOT_DISTANCE'] = ((df['LOC_X']**2 + df['LOC_Y']**2) ** 0.5 / 10).round().astype(int)
            # ===============================================
            
            return df
        except Exception:
            return pd.DataFrame()

    if tipo_temporada == "Regular Season + Playoffs":
        df_reg = buscar_por_tipo("Regular Season")
        df_play = buscar_por_tipo("Playoffs")
        df_completo = pd.concat([df_reg, df_play], ignore_index=True)
        return df_completo if not df_completo.empty else pd.DataFrame()
    else:
        return buscar_por_tipo(tipo_temporada)

@st.cache_data(ttl=3600)
def obter_elenco_time(nome_time, temporada, tipo="Regular Season"):
    """Cruza a tabela de temporada e jogadores para montar a tabela de rotação avançada (Ordenada por Minutos ou Pontos)."""
    abrev = TEAM_MAPPING.get(nome_time)
    if not abrev: return pd.DataFrame()
    
    try:
        q_trad = supabase.table("estatisticas_temporada").select("jogador_id, jogos_disputados, minutos_totais, pontos_totais, rebotes_totais, assistencias_totais").eq("temporada", temporada).eq("time_abrev", abrev)
        resp_trad = aplicar_filtro(q_trad, tipo).execute()
        
        if not resp_trad.data:
            from nba_api.stats.static import teams
            time_info = teams.find_team_by_abbreviation(abrev)
            if not time_info: return pd.DataFrame()
            
            resp_jog = supabase.table("jogadores").select("id, nome_completo, posicao").eq("time_atual_id", time_info['id']).execute()
            if not resp_jog.data: return pd.DataFrame()
            
            df_jog = pd.DataFrame(resp_jog.data).rename(columns={"id": "jogador_id"})
            for col in ['jogos_disputados', 'MPG', 'PPG', 'RPG', 'APG', 'ts_pct', 'usg_pct', 'net_rating']:
                df_jog[col] = 0.0
            
            ids = df_jog['jogador_id'].tolist()
            resp_past = supabase.table("estatisticas_temporada").select("jogador_id, minutos_totais, jogos_disputados").eq("temporada", "2025-26").in_("jogador_id", ids).execute()
            
            if resp_past.data:
                df_past = pd.DataFrame(resp_past.data)
                df_past = df_past[df_past['jogos_disputados'] > 0].copy()
                df_past['minutos_totais'] = pd.to_numeric(df_past.get('minutos_totais', 0), errors='coerce').fillna(0.0)
                df_past['MPG_past'] = (df_past['minutos_totais'] / df_past['jogos_disputados']).fillna(0.0)
                
                df_jog = pd.merge(df_jog, df_past[['jogador_id', 'MPG_past']], on="jogador_id", how="left").fillna(0)
                return df_jog.sort_values(by=["MPG_past", "PPG"], ascending=[False, False]).drop(columns=['MPG_past']).reset_index(drop=True)
                
            return df_jog

        q_adv = supabase.table("stats_avancadas").select("jogador_id, ts_pct, usg_pct, net_rating").eq("temporada", temporada)
        resp_adv = aplicar_filtro(q_adv, tipo).execute()
        
        df_trad = pd.DataFrame(resp_trad.data)
        df_adv = pd.DataFrame(resp_adv.data) if resp_adv.data else pd.DataFrame()
        
        ids_jogadores = df_trad['jogador_id'].tolist()
        resp_jog = supabase.table("jogadores").select("id, nome_completo, posicao").in_("id", ids_jogadores).execute()
        df_jog = pd.DataFrame(resp_jog.data) if resp_jog.data else pd.DataFrame()
        
        if not df_jog.empty:
            df_trad = df_trad.drop_duplicates(subset=['jogador_id'])
            df_adv = df_adv.drop_duplicates(subset=['jogador_id'])
            df_jog = df_jog.drop_duplicates(subset=['id'])

            df = pd.merge(df_trad, df_jog, left_on="jogador_id", right_on="id")
            if not df_adv.empty:
                df = pd.merge(df, df_adv, on="jogador_id", how="left")
                
            df = df[df['jogos_disputados'] > 0].copy()
            
            # TRATAMENTO ROBUSTO CONTRA NAN
            if 'minutos_totais' not in df.columns:
                df['minutos_totais'] = 0.0
            
            df['minutos_totais'] = pd.to_numeric(df['minutos_totais'], errors='coerce').fillna(0.0)
            df['MPG'] = (df['minutos_totais'] / df['jogos_disputados']).round(1).fillna(0.0)
            
            df['PPG'] = (df['pontos_totais'] / df['jogos_disputados']).round(1).fillna(0.0)
            df['RPG'] = (df['rebotes_totais'] / df['jogos_disputados']).round(1).fillna(0.0)
            df['APG'] = (df['assistencias_totais'] / df['jogos_disputados']).round(1).fillna(0.0)
            
            for col in ['ts_pct', 'usg_pct', 'net_rating']:
                if col not in df.columns: df[col] = 0.0
                df[col] = pd.to_numeric(df[col], errors='coerce').fillna(0.0)
                    
            # ORDENAÇÃO SEGURA: Primeiro por Minutos, depois por Pontos
            return df.sort_values(by=["MPG", "PPG"], ascending=[False, False]).reset_index(drop=True)
            
        return pd.DataFrame()
    except Exception as e: 
        print(f"Erro obter_elenco_time: {e}")
        return pd.DataFrame()
    
@st.cache_data(ttl=3600)
def obter_momento_recente(nome_time, temporada):
    """Busca os números dos últimos 10 jogos diretamente da NBA API, com segurança."""
    from nba_api.stats.endpoints import leaguedashplayerstats
    abrev = TEAM_MAPPING.get(nome_time)
    
    # Se a temporada for a atual (26-27) e ainda não tiver começado/não tiver 10 jogos, ou se o time não existir
    if not abrev or temporada == "2026-27":
        return None
        
    try:
        # Puxa a liga inteira nos últimos 10 jogos em 1 único request
        resp = leaguedashplayerstats.LeagueDashPlayerStats(
            season=temporada, 
            last_n_games=10,
            measure_type_detailed_defense="Base"
        ).get_data_frames()[0]
        
        # Filtra apenas o time selecionado
        df_time = resp[resp['TEAM_ABBREVIATION'] == abrev].copy()
        if df_time.empty: return None
        
        # Filtra quem jogou pelo menos 5 dos últimos 10 jogos para ser uma amostra válida
        df_time = df_time[df_time['GP'] >= 5].copy()
        if df_time.empty: return None
        
        # Calcula a média de pontos recente
        df_time['PPG_RECENTE'] = df_time['PTS'] / df_time['GP']
        
        # Encontra o jogador com maior pontuação no recorte
        destaque = df_time.sort_values(by="PPG_RECENTE", ascending=False).iloc[0]
        
        return {
            "nome": destaque['PLAYER_NAME'],
            "jogos": int(destaque['GP']),
            "ppg": round(destaque['PPG_RECENTE'], 1),
            "fg_pct": round(destaque['FG_PCT'] * 100, 1),
            "plus_minus": round(destaque['PLUS_MINUS'] / destaque['GP'], 1) # Saldo de pontos do time com ele em quadra
        }
    except Exception as e:
        return None # Se a API falhar ou não houver dados, retorna None para a página não quebrar


@st.cache_data(ttl=3600)
def obter_benchmarking_jogador(jogador_id, temporada, tipo="Regular Season"):
    """Calcula o percentil e a classificação do jogador em relação ao resto da liga."""
    df_liga = obter_dados_completos_liga(temporada, tipo)
    if df_liga.empty: return None

    # Filtro: Ignora quem jogou menos de 10 jogos para não distorcer as médias da liga
    df_valida = df_liga[df_liga['jogos_disputados'] >= 10].copy()
    if df_valida.empty or jogador_id not in df_valida['jogador_id'].values: return None

    jogador_stats = df_valida[df_valida['jogador_id'] == jogador_id].iloc[0]
    
    benchmarks = {}
    metricas = {
        'PPG': ('Pontos', False), 'RPG': ('Rebotes', False), 'APG': ('Assistências', False), 
        'ts_pct': ('True Shooting', True), 'usg_pct': ('Usage Rate', True), 
        'net_rating': ('Net Rating', False), 'pie': ('PIE', True), 'ast_pct': ('AST%', True)
    }
    
    for coluna, (nome_amigavel, is_pct) in metricas.items():
        if coluna not in df_valida.columns: continue
        
        valor = jogador_stats[coluna]
        media_liga = df_valida[coluna].mean()
        percentil = (df_valida[coluna] < valor).mean() * 100
        
        if percentil >= 90: classif, cor = "🌟 Elite", "normal"
        elif percentil >= 75: classif, cor = "📈 Acima Média", "normal"
        elif percentil >= 40: classif, cor = "📊 Na Média", "off"
        else: classif, cor = "📉 Abaixo Média", "inverse"
            
        benchmarks[coluna] = {
            "nome": nome_amigavel,
            "valor": valor * 100 if is_pct else valor,
            "media_liga": media_liga * 100 if is_pct else media_liga,
            "percentil": percentil,
            "classificacao": classif,
            "cor": cor,
            "is_pct": is_pct
        }
    return benchmarks

@st.cache_data(ttl=3600)
def obter_hot_streaks_liga(temporada="2026-27"):
    """Busca o desempenho da liga inteira nos últimos 10 jogos (Base e Avançado)."""
    from nba_api.stats.endpoints import leaguedashplayerstats
    import pandas as pd
    
    try:
        # Puxa estatísticas tradicionais (Pontos, Roubos, Tocos)
        df_base = leaguedashplayerstats.LeagueDashPlayerStats(
            season=temporada, last_n_games=10, measure_type_detailed_defense="Base"
        ).get_data_frames()[0]
        
        # Puxa estatísticas avançadas (TS%, Net Rating, PIE)
        df_adv = leaguedashplayerstats.LeagueDashPlayerStats(
            season=temporada, last_n_games=10, measure_type_detailed_defense="Advanced"
        ).get_data_frames()[0]
        
        if df_base.empty or df_adv.empty: return pd.DataFrame()
        
        # Junta tudo usando o ID do jogador
        df_completo = pd.merge(
            df_base[['PLAYER_ID', 'PLAYER_NAME', 'TEAM_ABBREVIATION', 'GP', 'MIN', 'PTS', 'STL', 'BLK']], 
            df_adv[['PLAYER_ID', 'TS_PCT', 'USG_PCT', 'NET_RATING', 'PIE']], 
            on='PLAYER_ID'
        )
        
        # Filtro de relevância: Apenas quem jogou pelo menos 5 dos últimos 10 jogos e mais de 15 min/jogo
        df_valido = df_completo[(df_completo['GP'] >= 5) & (df_completo['MIN'] >= 15)].copy()
        
        # Normaliza para médias por jogo
        df_valido['PPG'] = df_valido['PTS'] / df_valido['GP']
        df_valido['SPG'] = df_valido['STL'] / df_valido['GP']
        df_valido['BPG'] = df_valido['BLK'] / df_valido['GP']
        
        return df_valido
    except Exception as e:
        print(f"Erro ao buscar Hot Streaks: {e}")
        return pd.DataFrame()

        col4.metric("Defensive Rating", def_rating, help="Pontos sofridos a cada 100 posses em quadra.")
        
        st.divider()
        st.subheader("💡 Conclusões de Esforço e Marcação:")
        if deflections >= 3.0: st.success("Este jogador é um **pesadelo nas linhas de passe** (Elite em deflections).")

import streamlit as st
import pandas as pd

lista_jogadores = obter_lista_jogadores()
temporadas = ["2026-27", "2025-26", "2024-25", "2023-24", "2022-23"]
tipos_temporada = ["Regular Season", "Playoffs", "Regular Season + Playoffs"]

def renderizar_seletor(num_card):
    with st.container(border=True):
        col1, col2, col3 = st.columns(3)
        jog = col1.selectbox(f"Pesquisar Atleta {num_card}", ["Nenhum"] + lista_jogadores, key=f"jog_{num_card}")
        if jog == "Nenhum": return None
        
        temp = col2.selectbox(f"Temporada", temporadas, key=f"temp_{num_card}")
        tipo = col3.selectbox(f"Torneio", tipos_temporada, key=f"tipo_{num_card}")
        
        perfil = obter_perfil_jogador(jog)
        stats = obter_estatisticas_resumo(perfil['id'], temp, tipo) if perfil else None
        if stats: return {"nome": f"{jog} ({temp[-2:]})", "stats": stats, "perfil": perfil}
        return None

# Divulgação Progressiva
atleta1 = renderizar_seletor(1)
atleta2 = renderizar_seletor(2) if atleta1 else None
atleta3 = renderizar_seletor(3) if atleta2 else None

atletas_validos = [a for a in [atleta1, atleta2, atleta3] if a is not None]

if len(atletas_validos) > 0:
    st.divider()
    tabela_comparativa = []
    for a in atletas_validos:
        stats, jogos = a["stats"], max(a["stats"].get("jogos_disputados", 1), 1)
        tabela_comparativa.append({
            "Foto": f"https://cdn.nba.com/headshots/nba/latest/1040x760/{a['perfil']['id']}.png",

def classificar_arquetipo(stats, def_stats):
    """
    Analisa os dados de ataque, defesa e eficiência para classificar 
    o jogador em um arquétipo dinâmico da NBA moderna.
    A ordem importa: os mais dominantes/raros estão no topo.
    """
    if not stats:
        return None

    # Normalizar dados (Per Game e Porcentagens)
    jogos = stats.get('jogos_disputados', 1)
    if jogos == 0: jogos = 1
    
    ppg = stats.get('pontos_totais', 0) / jogos
    rpg = stats.get('rebotes_totais', 0) / jogos
    apg = stats.get('assistencias_totais', 0) / jogos
    
    usg = (stats.get('usg_pct') or 0) * 100
    ts = (stats.get('ts_pct') or 0) * 100
    ast_pct = (stats.get('ast_pct') or 0) * 100
    net_rtg = stats.get('net_rating') or 0
    
    fg3a = stats.get('fg3a_totais', 0) / jogos
    fg3m = stats.get('fg3m_totais', 0) / jogos
    pct_3pt = (fg3m / fg3a * 100) if fg3a > 0 else 0

    spg = def_stats.get('roubos_totais', 0) / jogos if def_stats else 0
    bpg = def_stats.get('tocos_totais', 0) / jogos if def_stats else 0
    defl = def_stats.get('deflections', 0) / jogos if def_stats else 0

    # ==========================================
    # TIER 1: SUPERSTARS E MOTORES OFENSIVOS
    # ==========================================
    
# 1A. Franchise Playmaker (Luka Doncic, Nikola Jokic)
    if ppg >= 24 and usg >= 28 and apg >= 7.5:
        return {
            "nome": "Franchise Playmaker", "icone": "🪄", 
            "cor": "linear-gradient(135deg, #FFD700, #FDB931)", 
            "desc": "O sistema ofensivo inteiro. Carrega a pontuação em altíssimo volume e atua simultaneamente como o principal maestro da equipe."
        }

    # 1B. Franchise Anchor (Joel Embiid, Giannis Antetokounmpo)
    if ppg >= 25 and usg >= 28 and rpg >= 10:
        return {
            "nome": "Franchise Anchor", "icone": "🦍", 
            "cor": "linear-gradient(135deg, #FFD700, #FDB931)", 
            "desc": "Força física e técnica imparável. Domina as tábuas de rebote, atrai marcação dupla constante e castiga defesas como a âncora da franquia."
        }

    # 1C. Franchise Scorer (Shai Gilgeous-Alexander, Kevin Durant, Jayson Tatum)
    if ppg >= 26 and usg >= 28:
        return {
            "nome": "Franchise Scorer", "icone": "👑", 
            "cor": "linear-gradient(135deg, #FFD700, #FDB931)", 
            "desc": "A primeira opção ofensiva indiscutível. Um pontuador letal, capaz de criar o próprio arremesso e destruir defesas em qualquer nível da quadra."
        }

    # 2. Offensive Engine (LaMelo Ball, Trae Young, Haliburton)
    if ppg >= 20 and apg >= 7:
        return {
            "nome": "Offensive Engine", "icone": "🏎️", 
            "cor": "linear-gradient(135deg, #FF512F, #DD2476)", 
            "desc": "O motor do ataque. Gera um volume absurdo de pontos por conta própria e criando para os companheiros."
        }

    # 3. Two-Way Star (Tatum, SGA, Edwards)

import pandas as pd

def calcular_percentil(valor_jogador, lista_todos_valores):
    """Calcula em qual percentil (0 a 100) o jogador está comparado ao resto da liga."""
    if valor_jogador is None or valor_jogador == "N/A" or not lista_todos_valores:
        return None
        
    # Limpa valores vazios da liga
    lista_limpa = [v for v in lista_todos_valores if v is not None]
    if not lista_limpa:
        return None
        
    serie = pd.Series(lista_limpa)
    # Calcula a porcentagem de jogadores que estão abaixo dele
    percentil = round((serie < valor_jogador).mean() * 100)
    return percentil

def gerar_texto_snapshot(usg_percentil, ts_percentil):
    """Gera o texto inteligente baseado no volume (USG%) e eficiência (TS%)."""
    if usg_percentil is None or ts_percentil is None:
        return "Dados insuficientes para gerar a análise de perfil deste jogador."
        
    # Análise de Volume (USG%)
    secao_volume = ""
    if usg_percentil >= 85:
        secao_volume = f"apresenta um **altíssimo volume ofensivo** (USG% no percentil {usg_percentil}), sendo uma das principais opções de ataque da equipe"
    elif usg_percentil >= 65:
        secao_volume = f"possui um **volume ofensivo relevante** (USG% no percentil {usg_percentil}), assumindo boa parte da criação do time"
    elif usg_percentil >= 35:
        secao_volume = f"tem um **papel ofensivo complementar** (USG% no percentil {usg_percentil}), atuando mais dentro do fluxo do sistema"
    else:
        secao_volume = f"possui **baixo volume de finalização** (USG% no percentil {usg_percentil}), focado em funções específicas ou finalizações esporádicas"

    # Análise de Eficiência (TS%)
    secao_eficiencia = ""
    if ts_percentil >= 80:
        secao_eficiencia = f"mantendo uma **eficiência de elite** na pontuação (TS% no percentil {ts_percentil})."
    elif ts_percentil >= 50:
        secao_eficiencia = f"com uma **eficiência acima da média** da liga (TS% no percentil {ts_percentil})."
    elif ts_percentil >= 30:
        secao_eficiencia = f"mas com uma **eficiência abaixo da média** em seus arremessos (TS% no percentil {ts_percentil})."
    else:
        secao_eficiencia = f"sofrendo com **baixa eficiência** na conversão de pontos (TS% no percentil {ts_percentil})."

    return f"Este jogador {secao_volume}, {secao_eficiencia}"

def gerar_tabela_comparativa(stats_jogador, df_liga):
    """Gera o DataFrame comparativo Jogador vs Liga com Percentis formatados."""
    if df_liga.empty or not stats_jogador:
        return pd.DataFrame()

    # Define as métricas que vamos comparar e as colunas correspondentes no df_liga
    metricas = [
        {"nome": "Pontos (PPG)", "coluna": "PPG", "formato": "{:.1f}"},
        {"nome": "Rebotes (RPG)", "coluna": "RPG", "formato": "{:.1f}"},
        {"nome": "Assistências (APG)", "coluna": "APG", "formato": "{:.1f}"},
        {"nome": "True Shooting (TS%)", "coluna": "ts_pct", "formato": "{:.1%}"},
        {"nome": "Usage Rate (USG%)", "coluna": "usg_pct", "formato": "{:.1%}"},
        {"nome": "Net Rating", "coluna": "net_rating", "formato": "{:.1f}"}
    ]
    
    tabela = []
    
    # Vamos precisar calcular o PPG, RPG, APG do jogador de novo aqui rapidinho
    jogos = stats_jogador.get("jogos_disputados", 1)
    valores_jogador = {
        "PPG": stats_jogador.get("pontos_totais", 0) / jogos,
        "RPG": stats_jogador.get("rebotes_totais", 0) / jogos,
        "APG": stats_jogador.get("assistencias_totais", 0) / jogos,
        "ts_pct": stats_jogador.get("ts_pct", 0),
        "usg_pct": stats_jogador.get("usg_pct", 0),
        "net_rating": stats_jogador.get("net_rating", 0)
    }

    for m in metricas:
        col = m["coluna"]
        val_jogador = valores_jogador[col]
        
        # Média da liga
        media_liga = df_liga[col].mean()
        
        # Percentil
        percentil = calcular_percentil(val_jogador, df_liga[col].tolist())
        
        # Formatação bonitinha
        fmt = m["formato"]
        
        tabela.append({
            "Métrica": m["nome"],
            "Jogador": fmt.format(val_jogador) if val_jogador is not None else "N/A",
            "Média da Liga": fmt.format(media_liga),
            "Percentil": f"{percentil}º" if percentil else "N/A"
        })
        
    return pd.DataFrame(tabela)

from nba_api.stats.static import teams

def obter_url_logo_time(identifier):
    """Retorna a URL oficial do logo da NBA baseado no ID numérico ou na sigla (ex: LAL)."""
    all_teams = teams.get_teams()
    team_obj = None
    
    if identifier is None:
        return None
        
    if str(identifier).isdigit():
        team_obj = next((t for t in all_teams if t['id'] == int(identifier)), None)
    else:
        identifier_str = str(identifier).upper()
        team_obj = next((t for t in all_teams if t['abbreviation'] == identifier_str), None)
        
    if team_obj:
        team_id = team_obj['id']
        return f"https://cdn.nba.com/logos/nba/{team_id}/global/L/logo.svg"
    return None

import urllib.request
import xml.etree.ElementTree as ET

def obter_noticias_jogador(nome_jogador):
    """Busca as últimas notícias do jogador na web usando RSS público do Google News."""
    try:
        query = nome_jogador.replace(" ", "+")
        url = f"https://news.google.com/rss/search?q={query}+NBA&hl=en-US&gl=US&ceid=US:en"
        
        req = urllib.request.Request(url, headers={'User-Agent': 'Mozilla/5.0'})
        with urllib.request.urlopen(req, timeout=3) as response:
            xml_data = response.read()
        
        root = ET.fromstring(xml_data)
        noticias = []
        
        # Coleta até 4 notícias mais recentes
        for item in root.findall('./channel/item')[:4]:
            titulo = item.find('title').text if item.find('title') is not None else ""
            link = item.find('link').text if item.find('link') is not None else "#"
            data = item.find('pubDate').text if item.find('pubDate') is not None else ""
            
            # Formata a data para ficar mais limpa (ex: Sat, 03 Oct 2026)
            data_limpa = data[:16] if len(data) >= 16 else data
            
            noticias.append({
                "titulo": titulo,
                "link": link,
                "data": data_limpa
            })
            
        return noticias
    except Exception:
        return []
